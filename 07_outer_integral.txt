# ============================================================
# Program 07 — Outer Integral I_out for ν = 2
# ============================================================
from sage.all import *
import numpy as np

def gamma_out(K, p, nu):
    N = (3*K)//40; h = (37*K)//40
    v = K - p*(K//p)
    u = max(0, N + v - p + 1)
    t_p = min(N, v) + u
    r_p = min(h, max(0, 2*h - K + nu*N - 2*p + 2))
    if K < 2*p:
        return -9*(K-p) + nu*t_p - 1 - min(r_p, p-1-N+u)
    else:
        return -9*(K-p) + 3 + 2*nu*N + (nu-1)*t_p - min(r_p, p+u)

def sum_floor_2i(K, p):
    h = (37*K)//40; n = h - 1
    if n < 1: return 0
    m = (2*n)//p
    return sum(n - (k*p + 1)//2 for k in range(1, m+1))

def vp_SK_outer(K, p, nu):
    N = (3*K)//40; h = (37*K)//40
    return 2*h*(K//p) - 2*nu*h*(N//p) - 2*sum_floor_2i(K, p)

def T_of_y(y, K, nu):
    p = int(round(y*K))
    if p <= 0 or 3*p <= K or p > K:
        return None
    return -(vp_SK_outer(K, p, nu) + gamma_out(K, p, nu)) / K

K = 40 * 10**6
nu = 2

print("=" * 62)
print("Program 07 — Outer Integral I_out for ν = 2")
print("=" * 62)
print(f"ν = {nu},  K = {K}")
print("-" * 62)

# Integrate T(y) on [1/3, 1]
n_pts = 4001
ys = np.linspace(1/3 + 1e-7, 1.0 - 1e-7, n_pts)
d_y = ys[1] - ys[0]
Ts = np.array([float(T_of_y(y, K, nu) or 0.0) for y in ys])

# Simpson's rule
I_outer = d_y/3 * (Ts[0] + Ts[-1]
                   + 4*Ts[1:-1:2].sum()
                   + 2*Ts[2:-2:2].sum())

# Scalar extra on [1, 2λ]: ∫ ∑_{j=1}^{ν-1}(2λ-jy)_+ dy
lam = 37.0/40
two_lam = 2*lam
extra = 0.0
for j in range(1, nu):
    y0 = 1.0
    y1 = min(two_lam, two_lam/j)
    if y1 > y0:
        extra += two_lam*(y1-y0) - j*(y1**2 - y0**2)/2

I_out = I_outer + extra

print(f"I_outer on [1/3, 1]     = {I_outer:.10f}")
print(f"Extra scalar on [1, 2λ] = {extra:.10f}")
print(f"I_out (total)           = {I_out:.10f}")
print("-" * 62)

# Sample T(y) values
print("Diagnostic T(y) values:")
print(f"{'y':>8}  {'T(y)':>14}")
for y in [0.34, 0.40, 0.50, 0.55, 0.60, 0.65, 0.70, 0.75,
          0.80, 0.85, 0.90, 0.95, 1.00]:
    t = T_of_y(y, K, nu)
    if t is not None:
        print(f"{y:>8.3f}  {float(t):>14.6f}")
print("-" * 62)

# Sanity checks
check_finite = np.isfinite(I_out)
check_positive = I_out > 0
print(f"Sanity checks:")
print(f"  Finite:        {check_finite}")
print(f"  I_out > 0:     {check_positive}")
ok = check_finite and check_positive
print(f"==> {'PASS' if ok else 'FAIL'} (Program 07)")
print("=" * 62)

print()
print("Copy this for downstream use:")
print(f"  I_out = {I_out:.10f}")
